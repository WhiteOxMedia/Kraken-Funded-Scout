=== Install Bridge.bat ===  
@echo off
REM ============================================================
REM  Install Scout-to-Bot Signal Bridge
REM  Copies signal_bridge.py and scout_signal_reader.py to the
REM  kraken_xrp_trading_bot project and patches bot_engine.py
REM ============================================================

cd /d "%~dp0"

echo Installing signal bridge to kraken_xrp_trading_bot...

REM Copy the bridge files
copy "signal_bridge.py" "C:\Users\birds\Workspace\kraken_xrp_trading_bot\signal_bridge.py"
copy "scout_signal_reader.py" "C:\Users\birds\Workspace\kraken_xrp_trading_bot\scout_signal_reader.py"

echo.
echo Files copied. Now you need to patch bot_engine.py manually:
echo.
echo 1. Open C:\Users\birds\Workspace\kraken_xrp_trading_bot\bot_engine.py
echo 2. Find the line:  def run_tick(self)
echo 3. Add these lines right after it (before the try:):
echo.
echo     from scout_signal_reader import check_scout_signals
echo     scout_result = check_scout_signals(self.kraken, self.config)
echo     if scout_result:
echo         return scout_result
echo.
echo 4. Save the file
echo 5. Restart the XRP bot
echo.
echo The scout will now write buy/sell signals to trade_signals.json
echo The XRP bot will read and execute them on your real Kraken account.
echo.
pause
=== Kill Switch - FLATTEN.bat ===  
@echo off
title Kraken Scout - Kill Switch (FLATTEN ALL)
cd /d "%~dp0"

if exist "kill_switch.lock" (
    echo ================================================================
    echo   Kill switch is ALREADY ACTIVE
    echo   All trades are blocked. Positions should be flattened.
    echo ================================================================
    echo.
    echo   To turn off: Run 'Kill Switch - RESET.bat'
    echo.
    pause
    exit /b
)

echo ================================================================
echo   KILL SWITCH - FLATTEN ALL POSITIONS
echo ================================================================
echo.
echo   This will:
echo     1. Activate the kill switch (blocks all new trades)
echo     2. Sell ALL positions on Kraken at market
echo     3. Clear the signal bridge database
echo.
echo   The scanner and bot must be running for this to work.
echo   The kill switch stays active until you reset it.
echo.
echo   Press any key to FLATTEN, or close this window to cancel.
pause >nul

echo.
echo   Activating kill switch...
set "PY=.venv\Scripts\python.exe"
%PY% --version >nul 2>&1
if errorlevel 1 set "PY=C:\Python314\python.exe"
%PY% -c "from signal_bridge import trigger_kill_switch; trigger_kill_switch('Manual kill switch - flatten all positions from desktop')"

echo.
echo ================================================================
echo   Kill switch activated. Check scanner/bot logs for sell output.
echo   Run 'Kill Switch - RESET.bat' to turn it off and resume trading.
echo ================================================================
echo.
pause
=== Kill Switch - RESET.bat ===  
@echo off
title Kraken Scout - Kill Switch RESET
cd /d "%~dp0"

if not exist "kill_switch.lock" (
    echo ================================================================
    echo   Kill switch is NOT active
    echo   Trading is running normally.
    echo ================================================================
    echo.
    pause
    exit /b
)

echo ================================================================
echo   KILL SWITCH - RESET (RESUME TRADING)
echo ================================================================
echo.
echo   This will:
echo     1. Delete the kill switch lock file
echo     2. Resume normal trading (scanner + bot)
echo.
echo   Press any key to RESET, or close this window to cancel.

echo.
echo   Resetting kill switch...
set "PY=.venv\Scripts\python.exe"
%PY% --version >nul 2>&1
if errorlevel 1 set "PY=C:\Python314\python.exe"

echo.
echo ================================================================
echo   Kill switch reset. Trading resumes on next scan cycle.
echo ================================================================
echo.
pause
=== Launch Kraken.bat ===  
@echo off
title Kraken — Full Stack Launcher
cd /d "%~dp0"

echo ============================================================
echo   KRAKEN — launching full stack
echo   1. Volatility Engine   (thresholds from real data)
echo   2. Scout Scanner       (rotation-aware auto-scanner)
echo   3. Trader Bot          (executes bridge signals)
echo   4. Console Dashboard   (localhost:8501)
echo ============================================================

if not exist ".venv\Scripts\python.exe" (
    echo Setting up venv...
    uv sync
) else (
    ".venv\Scripts\python.exe" --version >nul 2>&1
    if errorlevel 1 (
        echo Venv is broken - rebuilding...
        rmdir /s /q .venv 2>nul
        uv sync
    )
)
if exist "__pycache__" rmdir /s /q "__pycache__"
if exist "trend\__pycache__" rmdir /s /q "trend\__pycache__"

REM ── 1. Volatility Engine (own window, dup-safe) ───────────────────────
REM (PowerShell checks command lines; tasklist/wmic do not work reliably)
powershell -NoProfile -Command "if (Get-CimInstance Win32_Process -Filter \"name='python.exe'\" | Where-Object {$_.CommandLine -like '*volatility_engine*'}) { exit 0 } else { exit 1 }" >nul 2>&1
if errorlevel 1 (
    start "Volatility Engine" cmd /c ".venv\Scripts\python.exe -X utf8 volatility_engine.py 2>&1 & pause"
    echo Volatility Engine started.
) else (
    echo Volatility Engine already running - skipping.
)

REM ── 2. Scanner (own window, dup-safe) ─────────────────────────────────
powershell -NoProfile -Command "if (Get-CimInstance Win32_Process -Filter \"name='python.exe'\" | Where-Object {$_.CommandLine -like '*background_scanner*'}) { exit 0 } else { exit 1 }" >nul 2>&1
if errorlevel 1 (
    start "Scout Scanner" cmd /c ".venv\Scripts\python.exe background_scanner.py 2>&1 & pause"
    echo Scanner started.
) else (
    echo Scanner already running - skipping.
)

REM ── 3. Trader bot (own window, dup-safe, uses the BOT venv) ───────────
set BOT_DIR=C:\Users\birds\Workspace\kraken_xrp_trading_bot
powershell -NoProfile -Command "if (Get-CimInstance Win32_Process -Filter \"name='python.exe'\" | Where-Object {$_.CommandLine -like '*bot_runner*'}) { exit 0 } else { exit 1 }" >nul 2>&1
if errorlevel 1 (
    start "Kraken Trader" cmd /c "cd /d %BOT_DIR% && .venv\Scripts\python.exe bot_runner.py --interval 15 2>&1 & pause"
    echo Trader bot started.
) else (
    echo Trader bot already running - skipping.
)

REM ── 4. Console dashboard (this window; open browser if already up) ────
ping -n 4 127.0.0.1 >nul
netstat -an | findstr ":8501" | findstr "LISTENING" >nul 2>&1
if not errorlevel 1 (
    echo Console already running - opening browser...
    start "" "http://localhost:8501"
    ping -n 3 127.0.0.1 >nul
    exit /b
)

echo Starting console on http://localhost:8501...
".venv\Scripts\streamlit.exe" run app.py --server.port 8501 --server.runOnSave false

echo.
echo Dashboard stopped. Scanner / Volatility Engine / Trader run in separate windows.
pause
=== Launch Scout.bat ===  
@echo off
title Kraken Scout - Scanner + Dashboard
cd /d "%~dp0"

if not exist ".venv\Scripts\python.exe" (
    echo Setting up venv...
    uv sync
) else (
    ".venv\Scripts\python.exe" --version >nul 2>&1
    if errorlevel 1 (
        echo Venv is broken - rebuilding...
        rmdir /s /q .venv 2>nul
        uv sync
    )
)
if exist "__pycache__" rmdir /s /q "__pycache__"
if exist "trend\__pycache__" rmdir /s /q "trend\__pycache__"

REM Check if already running - don't start duplicate scanner
REM (PowerShell checks command lines; tasklist/wmic do not work reliably)
powershell -NoProfile -Command "if (Get-CimInstance Win32_Process -Filter \"name='python.exe'\" | Where-Object {$_.CommandLine -like '*background_scanner*'}) { exit 0 } else { exit 1 }" >nul 2>&1
if errorlevel 1 (
    REM Scanner NOT running - start it in a new window
    start "Scout Scanner" cmd /c ".venv\Scripts\python.exe background_scanner.py 2>&1 & pause"
) else (
    echo Scanner is already running - skipping scanner start.
)

ping -n 3 127.0.0.1 >nul

REM If the console dashboard is already serving, just open the browser and exit
netstat -an | findstr ":8501" | findstr "LISTENING" >nul 2>&1
if not errorlevel 1 (
    echo Console already running - opening browser...
    start "" "http://localhost:8501"
    ping -n 3 127.0.0.1 >nul
    exit /b
)

REM Start dashboard in this window - opens the browser automatically
echo Starting console on http://localhost:8501...
".venv\Scripts\streamlit.exe" run app.py --server.port 8501 --server.runOnSave false

echo.
echo Dashboard stopped. Scanner is in a separate window.
pause
